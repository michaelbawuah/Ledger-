cmake_minimum_required(VERSION 3.20)
project(Ledger VERSION 1.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(LEDGER_NATIVE "Enable host-specific ISA for local benchmarks" OFF)
option(LEDGER_FUZZ "Build Clang libFuzzer parser target" OFF)
option(LEDGER_REQUIRE_GTEST "Fail if GoogleTest is unavailable" OFF)
set(LEDGER_SANITIZER "" CACHE STRING "address or thread")
find_package(Threads REQUIRED)
if(MSVC)
  add_compile_options(/W4 /permissive-)
else()
  add_compile_options(-Wall -Wextra -Wpedantic)
  if(LEDGER_NATIVE)
    add_compile_options(-march=native)
  endif()
  if(LEDGER_SANITIZER STREQUAL "address")
    add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
    add_link_options(-fsanitize=address,undefined)
  elseif(LEDGER_SANITIZER STREQUAL "thread")
    add_compile_options(-fsanitize=thread -fno-omit-frame-pointer)
    add_link_options(-fsanitize=thread)
  endif()
endif()

add_library(ledger src/book.cpp src/reference_book.cpp src/feed.cpp)
target_include_directories(ledger PUBLIC include)
target_link_libraries(ledger PUBLIC Threads::Threads)
foreach(pair IN ITEMS "ledger_replay;src/replay.cpp" "ledger_generate;bench/gen_orders.cpp" "ledger_bench;bench/bench_replay.cpp" "ledger_spsc_bench;bench/bench_spsc.cpp" "ledger_market_maker;src/market_maker.cpp")
  list(GET pair 0 target)
  list(GET pair 1 source)
  add_executable(${target} ${source})
  target_link_libraries(${target} PRIVATE ledger)
endforeach()

include(CTest)
if(BUILD_TESTING)
  find_package(GTest QUIET)
  if(LEDGER_REQUIRE_GTEST AND NOT GTest_FOUND)
    message(FATAL_ERROR "Install GoogleTest or use the bundled test harness with LEDGER_REQUIRE_GTEST=OFF")
  endif()
  add_executable(ledger_tests tests/unit/engine_tests.cpp)
  target_link_libraries(ledger_tests PRIVATE ledger)
  if(GTest_FOUND)
    target_link_libraries(ledger_tests PRIVATE GTest::gtest)
    target_compile_definitions(ledger_tests PRIVATE LEDGER_USE_GTEST=1)
  else()
    message(STATUS "GoogleTest unavailable: using the same cases in the dependency-free harness")
  endif()
  add_executable(ledger_allocation_tests tests/unit/memory_tests.cpp)
  target_link_libraries(ledger_allocation_tests PRIVATE ledger)
  add_executable(ledger_spsc_tests tests/unit/spsc_tests.cpp)
  target_link_libraries(ledger_spsc_tests PRIVATE ledger)
  add_executable(ledger_feed_tests tests/unit/feed_tests.cpp)
  target_link_libraries(ledger_feed_tests PRIVATE ledger)
  add_executable(ledger_differential tests/differential/differential.cpp)
  target_link_libraries(ledger_differential PRIVATE ledger)
  add_test(NAME engine_semantics COMMAND ledger_tests)
  add_test(NAME allocation_free COMMAND ledger_allocation_tests)
  add_test(NAME spsc_stress COMMAND ledger_spsc_tests)
  add_test(NAME feed_roundtrip COMMAND ledger_feed_tests)
  add_test(NAME differential_smoke COMMAND ledger_differential --seeds 5 --orders 10000)
  set_tests_properties(spsc_stress differential_smoke PROPERTIES TIMEOUT 120)
endif()

if(LEDGER_FUZZ)
  if(NOT CMAKE_CXX_COMPILER_ID MATCHES "Clang")
    message(FATAL_ERROR "LEDGER_FUZZ requires Clang")
  endif()
  add_executable(ledger_parser_fuzz tests/fuzz/parser_fuzz.cpp)
  target_link_libraries(ledger_parser_fuzz PRIVATE ledger)
  target_compile_options(ledger PRIVATE -fsanitize=fuzzer-no-link,address,undefined)
  target_compile_options(ledger_parser_fuzz PRIVATE -fsanitize=fuzzer,address,undefined)
  target_link_options(ledger_parser_fuzz PRIVATE -fsanitize=fuzzer,address,undefined)
endif()
